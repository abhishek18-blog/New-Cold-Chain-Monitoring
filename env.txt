# ------------------------------------------------------------------------------
# OPTION 1: PRODUCTION GCP CLOUD SHELL SETUP (RECOMMENDED FOR GCLOUD CLI)
# ------------------------------------------------------------------------------
# Run this in your Google Cloud Shell terminal. It automatically detects your
# active GCP project ID and creates the production .env file:


# 1. Resolve the GCP Project ID first safely
GCP_PROJECT_ID=$(gcloud config get-value project 2>/dev/null)
if [ -z "$GCP_PROJECT_ID" ] || [ "$GCP_PROJECT_ID" = "(unset)" ]; then
    GCP_PROJECT_ID="cloud-cold-chain"
fi

# 2. Write the .env file directly with the resolved variable
cat << EOF > .env
# Auto-detected Google Cloud Project
GCP_PROJECT_ID=$GCP_PROJECT_ID

# Pub/Sub Topics & Subscriptions
PUBSUB_TOPIC=telemetry-topic
PUBSUB_SUB=telemetry-sub
PUBSUB_DLQ_TOPIC=telemetry-dlq-topic
PUBSUB_ALERTS_TOPIC=alerts-topic
PUBSUB_ALERTS_SUB=alerts-sub

# Telemetry Generator Settings (1.0s interval protecting Firestore quotas)
PING_INTERVAL_SEC=1.0
EXCURSION_PROBABILITY=0.015

# Driver Real-Time Alerting Network (Discord Webhook)
DISCORD_WEBHOOK_URL="https://discord.com/api/webhooks/1563536821294469202/JNHwpAjZGGCxKOK_k6a93Bsk_7rvWQsk15gbgqFob6pQEW3ESzRyWdXimAJiTFIZmVKn"
DISCORD_VOICE_TTS=false
ALERT_COOLDOWN_SEC=300

# Silence / Signal Loss Detector
SILENCE_TIMEOUT_SEC=120
SILENCE_CHECK_INTERVAL_SEC=30

# Web Dashboard Port
DASHBOARD_PORT=8000
EOF

echo "✅ Created production .env file for GCP!"
cat .env
